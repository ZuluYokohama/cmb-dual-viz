cmake_minimum_required(VERSION 3.20)
project(cmb_llama_observer LANGUAGES C CXX)
set(LLAMA_SOURCE_DIR "" CACHE PATH "Existing llama.cpp checkout")
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
  message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to a llama.cpp checkout")
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${LLAMA_SOURCE_DIR}"
  OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LLAMA_REV STREQUAL "7ab4ee7baad2d920464cbacfad4f4b07cf111fd2")
  message(FATAL_ERROR "Observer v1 requires the documented llama.cpp revision")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama-build)
add_executable(cmb-llama-observer main.cpp)
target_compile_features(cmb-llama-observer PRIVATE cxx_std_17)
target_compile_definitions(cmb-llama-observer PRIVATE CMB_LLAMA_REV="${LLAMA_REV}")
target_include_directories(cmb-llama-observer PRIVATE "${LLAMA_SOURCE_DIR}/vendor")
target_link_libraries(cmb-llama-observer PRIVATE llama)
enable_testing()
add_executable(cmb-metrics-test metrics_test.cpp)
target_compile_features(cmb-metrics-test PRIVATE cxx_std_17)
add_test(NAME cmb-metrics COMMAND cmb-metrics-test)
